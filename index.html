คู่มือเชื่อม Google Sheets

เปิดแต่ละหัวข้อเพื่อดูขั้นตอน Deploy และโค้ด Google Apps Script ที่คัดลอกไปใช้งานได้

A) สร้าง Google Sheet และเปิด Apps Script
  1. สร้าง Google Sheet ใหม่สำหรับเก็บรายการจอง
  2. เลือกเมนู Extensions > Apps Script
  3. ลบโค้ดเดิมในโปรเจกต์ แล้ววางโค้ดในหัวข้อถัดไป
B) โค้ด Google Apps Script สำหรับรับข้อมูลการจอง

เปลี่ยน SPREADSHEET_ID เป็นรหัส Google Sheet ของคุณจาก URL ของชีต

function doPost(e) {
  try {
    const SPREADSHEET_ID = 'PUT_YOUR_SPREADSHEET_ID_HERE';
    const SHEET_NAME = 'Reservations';

    const payload = JSON.parse(e.postData.contents || '{}');
    const spreadsheet = SpreadsheetApp.openById(SPREADSHEET_ID);
    let sheet = spreadsheet.getSheetByName(SHEET_NAME);

    if (!sheet) {
      sheet = spreadsheet.insertSheet(SHEET_NAME);
    }

    const headers = [
      'booking_id', 'first_name', 'last_name', 'phone', 'email',
      'payment_method', 'seats', 'total_price', 'status', 'created_at'
    ];

    if (sheet.getLastRow() === 0) {
      sheet.appendRow(headers);
      sheet.setFrozenRows(1);
    }

    sheet.appendRow(headers.map(function(header) {
      return payload[header] || '';
    }));

    return ContentService
      .createTextOutput(JSON.stringify({
        success: true,
        message: 'บันทึกข้อมูลการจองเรียบร้อยแล้ว'
      }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {
    return ContentService
      .createTextOutput(JSON.stringify({
        success: false,
        error: error.message
      }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
C) Deploy เป็น Web app
  1. กด Deploy > New deployment
  2. เลือกประเภท Web app
  3. ตั้งค่า Execute as: Me
  4. ตั้งค่า Who has access เป็นตัวเลือกที่อนุญาตให้ผู้ใช้แอปส่งข้อมูลได้ เช่น Anyone หรือ Anyone with Google account ตามนโยบายของคุณ
  5. กด Deploy อนุญาตสิทธิ์ แล้วคัดลอก Web app URL ที่ลงท้ายด้วย /exec
D) วาง Web App URL และข้อควรระวังด้านความปลอดภัย

นำ Web app URL ไปวางในช่อง “Web App URL ของ Google Apps Script” ด้านบน แล้วกดบันทึก URL จากนั้นหลังยืนยันใบจอง ให้กดส่งข้อมูลเข้า Google Sheets ได้

ข้อควรระวัง: ห้ามใส่รหัสผ่าน API key ข้อมูลบัตร หรือข้อมูลลับใด ๆ ไว้ในหน้าเว็บหรือ Google Apps Script ที่เผยแพร่ให้ผู้อื่นใช้งาน

E) README และจุดตั้งค่าราคา/ผังที่นั่งในแอป

คู่มือนี้ทำหน้าที่เป็น README ในตัวแอปแล้ว ส่วน CONFIG ด้านบนของ JavaScript มีทั้งราคาเริ่มต้นและโครงสร้างข้อมูลผังที่นั่ง คุณสามารถแก้ไขและคัดลอกไปใช้งานต่อได้อย่างเป็นระบบ
